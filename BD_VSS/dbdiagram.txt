////////////////////////////////////////////////////////////
// TABLA: usuario
////////////////////////////////////////////////////////////
Table usuario {
  id bigint [pk, note: 'Identificador único']
  nombre varchar(100) [note: 'Nombre completo del usuario']
  correo varchar(100) [note: 'Correo electrónico']
  rol varchar(50) [note: 'Rol del usuario']
  activo boolean [note: 'Estado activo/inactivo']
  fecha_creacion timestamp [note: 'Fecha de creación']
  password_hash varchar(255) [note: 'Contraseña encriptada']
}

////////////////////////////////////////////////////////////
// TABLA: tipo_cambio
////////////////////////////////////////////////////////////
Table tipo_cambio {
  id bigint [pk]
  fecha date [unique, note: 'Fecha del tipo de cambio']
  valor decimal(10,4) [note: 'Valor del dólar']
  fuente varchar(100) [note: 'Fuente del dato']
  usuario_registro_id bigint [ref: > usuario.id, note: 'Usuario que registró']
  fecha_registro timestamp
}

////////////////////////////////////////////////////////////
// TABLA: parametros_comerciales
////////////////////////////////////////////////////////////
Table parametros_comerciales {
  id bigint [pk]
  tipo_cambio_usado decimal(10,4) [note: 'Tipo de cambio positivo']
  porcentaje_utilidad decimal(5,2) [note: 'Margen válido']
  fecha_vigencia date [note: 'Fecha desde la cual aplica']
  usuario_editor_id bigint [ref: > usuario.id, note: 'Usuario que modificó']
  fecha_actualizacion timestamp [note: 'Fecha de actualización']
}

////////////////////////////////////////////////////////////
// TABLA: auditoria_parametros
////////////////////////////////////////////////////////////
Table auditoria_parametros {
  id bigint [pk]
  parametro varchar(100) [note: 'Nombre del parámetro']
  valor_anterior varchar(100)
  valor_nuevo varchar(100)
  usuario_id bigint [ref: > usuario.id]
  fecha timestamp
}

////////////////////////////////////////////////////////////
// TABLA: cliente
////////////////////////////////////////////////////////////
Table cliente {
  id bigint [pk]
  nombre_barco varchar(100) [unique, note: 'Nombre del barco']
  imo_numero varchar(20)
  empresa_operadora varchar(100)
  correo_contacto varchar(100)
  puerto_base varchar(100)
  pais_origen varchar(50)
  tipo_cliente varchar(50)
  capitán varchar(100)
  cocinero_jefe varchar(100)
  observaciones text
  activo boolean
  fecha_creacion timestamp
}

////////////////////////////////////////////////////////////
// TABLA: cotizacion
////////////////////////////////////////////////////////////
Table cotizacion {
  id bigint [pk]
  cliente_id bigint [ref: > cliente.id]
  fecha_solicitud date
  puerto_destino varchar(100)
  archivo_origen varchar(100)
  observaciones text
  usuario_importador_id bigint [ref: > usuario.id]
  fecha_importacion timestamp
}

////////////////////////////////////////////////////////////
// TABLA: familia_producto
////////////////////////////////////////////////////////////
Table familia_producto {
  id bigint [pk]
  nombre varchar(100) [unique]
  descripcion text
}

////////////////////////////////////////////////////////////
// TABLA: producto
////////////////////////////////////////////////////////////
Table producto {
  id bigint [pk]
  familia_id bigint [ref: > familia_producto.id]
  descripcion_en text
  descripcion_es text
  unidad_medida varchar(20)
  valor_pesos decimal(12,2)
  fecha_actualizacion timestamp
  usuario_editor_id bigint [ref: > usuario.id]
}

////////////////////////////////////////////////////////////
// TABLA: item_cotizacion
////////////////////////////////////////////////////////////
Table item_cotizacion {
  id bigint [pk]
  cotizacion_id bigint [ref: > cotizacion.id]
  producto_id bigint [ref: > producto.id]
  descripcion text
  unidad_medida varchar(20)
  valor_pesos decimal(12,2)
  tipo_cambio_usado decimal(10,4)
  precio_costo_neto decimal(12,2)
  porcentaje_utilidad decimal(5,2)
  precio_venta_neto decimal(12,2)
  cantidad int
  total_neto decimal(12,2)
  comentarios text
}

////////////////////////////////////////////////////////////
// TABLA: log_importacion
////////////////////////////////////////////////////////////
Table log_importacion {
  id bigint [pk]
  cotizacion_id bigint [ref: > cotizacion.id]
  usuario_id bigint [ref: > usuario.id]
  fecha timestamp
  total_items int
  items_vinculados int
  items_no_encontrados int
  errores_detectados text
  resumen text
}

////////////////////////////////////////////////////////////
// TABLA: brokers
////////////////////////////////////////////////////////////
Table brokers {
  broker_id int [pk, note: 'Identificador único']
  broker_name varchar(255) [note: 'Nombre del broker']
  descripcion text
  contacto varchar(255)
  email varchar(255)
  activo boolean
  fecha_creacion timestamp
  fecha_actualizacion timestamp
}

////////////////////////////////////////////////////////////
// TABLA: broker_formatos
////////////////////////////////////////////////////////////
Table broker_formatos {
  formato_id int [pk]
  broker_id int [ref: > brokers.broker_id]
  version varchar(50)
  header_row int
  descripcion text
  archivo_ejemplo varchar(500)
  activo boolean
  fecha_creacion timestamp
  fecha_actualizacion timestamp
}

////////////////////////////////////////////////////////////
// TABLA: formato_columnas
////////////////////////////////////////////////////////////
Table formato_columnas {
  columna_id int [pk]
  formato_id int [ref: > broker_formatos.formato_id]
  campo_estandar varchar(100)
  nombre_columna_original varchar(255)
  indice_columna int
  letra_columna varchar(10)
  tipo_dato varchar(50)
  requerido boolean
  descripcion text
  fecha_creacion timestamp
  color_fondo varchar(20)
  color_texto varchar(20)
  es_negrita boolean
  es_cursiva boolean
  tiene_borde boolean
}

////////////////////////////////////////////////////////////
// TABLA: broker_metadata
////////////////////////////////////////////////////////////
Table broker_metadata {
  metadata_id int [pk]
  formato_id int [ref: > broker_formatos.formato_id]
  seccion varchar(100)
  campo_nombre varchar(100)
  campo_valor text
  fila_origen int
  columna_origen int
  letra_columna varchar(5)
  fecha_creacion timestamp
}

////////////////////////////////////////////////////////////
// TABLA: cotizaciones_archivos
////////////////////////////////////////////////////////////
Table cotizaciones_archivos {
  archivo_id int [pk]
  broker_id int [ref: > brokers.broker_id]
  formato_id int [ref: > broker_formatos.formato_id]
  nombre_archivo varchar(500)
  ruta_archivo text
  vessel_name varchar(255)
  imo_number varchar(50)
  numero_cotizacion varchar(255)
  fecha_archivo date
  fecha_procesado timestamp
}

////////////////////////////////////////////////////////////
// TABLA: archivo_colores
////////////////////////////////////////////////////////////
Table archivo_colores {
  color_id int [pk]
  archivo_id int [ref: > cotizaciones_archivos.archivo_id]
  campo_estandar varchar(100)
  nombre_columna_original varchar(255)
  indice_columna int
  letra_columna varchar(10)
  color_fondo varchar(20)
  color_texto varchar(20)
  es_negrita boolean
  es_cursiva boolean
  tiene_borde boolean
  fecha_deteccion timestamp
}
