// ------------------------------------------------------------
// BLOQUE A — TABLAS DEL SISTEMA (MODELO DITTO)
// ------------------------------------------------------------

Table usuario {
  id bigint [pk, increment]
  nombre varchar(100)
  correo varchar(100)
  rol varchar(50)
  activo boolean
  fecha_creacion timestamp
  password_hash varchar(255)
}

Table tipo_cambio {
  id bigint [pk, increment]
  fecha date [unique]
  valor decimal(10,4)
  fuente varchar(100)
  usuario_registro_id bigint
  fecha_registro timestamp
}

Ref: tipo_cambio.usuario_registro_id > usuario.id

Table parametros_comerciales {
  id bigint [pk, increment]
  tipo_cambio_usado decimal(10,4)
  porcentaje_utilidad decimal(5,2)
  fecha_vigencia date
  usuario_editor_id bigint
  fecha_actualizacion timestamp
}

Ref: parametros_comerciales.usuario_editor_id > usuario.id

Table auditoria_parametros {
  id bigint [pk, increment]
  parametro varchar(100)
  valor_anterior varchar(100)
  valor_nuevo varchar(100)
  usuario_id bigint
  fecha timestamp
}

Ref: auditoria_parametros.usuario_id > usuario.id

Table cliente {
  id bigint [pk, increment]
  nombre_barco varchar(100)
  imo_numero varchar(20)
  empresa_operadora varchar(100)
  correo_contacto varchar(100)
  puerto_base varchar(100)
  pais_origen varchar(50)
  tipo_cliente varchar(50)
  capitán varchar(100)
  cocinero_jefe varchar(100)
  observaciones text
  activo boolean
  fecha_creacion timestamp
}

Table cotizacion {
  id bigint [pk, increment]
  cliente_id bigint
  fecha_solicitud date
  puerto_destino varchar(100)
  archivo_origen varchar(100)
  observaciones text
  usuario_importador_id bigint
  fecha_importacion timestamp
}

Ref: cotizacion.cliente_id > cliente.id
Ref: cotizacion.usuario_importador_id > usuario.id

Table familia_producto {
  id bigint [pk, increment]
  nombre varchar(100)
  descripcion text
}

Table producto {
  id bigint [pk, increment]
  familia_id bigint
  descripcion_en text
  descripcion_es text
  unidad_medida varchar(20)
  valor_pesos decimal(12,2)
  fecha_actualizacion timestamp
  usuario_editor_id bigint
}

Ref: producto.familia_id > familia_producto.id
Ref: producto.usuario_editor_id > usuario.id

// ------------------------------------------------------------
// NUEVA TABLA — SINÓNIMOS DE PRODUCTOS
// ------------------------------------------------------------

Table producto_sinonimo {
  id bigint [pk, increment]
  producto_id bigint
  nombre_sinonimo text
  fecha_registro timestamp
  usuario varchar(100)
}

Ref: producto_sinonimo.producto_id > producto.id

// ------------------------------------------------------------

Table item_cotizacion {
  id bigint [pk, increment]
  cotizacion_id bigint
  producto_id bigint
  descripcion text
  unidad_medida varchar(20)
  valor_pesos decimal(12,2)
  tipo_cambio_usado decimal(10,4)
  precio_costo_neto decimal(12,2)
  porcentaje_utilidad decimal(5,2)
  precio_venta_neto decimal(12,2)
  cantidad int
  total_neto decimal(12,2)
  comentarios text
}

Ref: item_cotizacion.cotizacion_id > cotizacion.id
Ref: item_cotizacion.producto_id > producto.id

Table log_importacion {
  id bigint [pk, increment]
  cotizacion_id bigint
  usuario_id bigint
  fecha timestamp
  total_items int
  items_vinculados int
  items_no_encontrados int
  errores_detectados text
  resumen text
}

Ref: log_importacion.cotizacion_id > cotizacion.id
Ref: log_importacion.usuario_id > usuario.id

// ------------------------------------------------------------
// BLOQUE B — MODELO TOBE
// ------------------------------------------------------------

Table brokers {
  broker_id int [pk, increment]
  broker_name varchar(255)
  descripcion text
  contacto varchar(255)
  email varchar(255)
  activo boolean
  fecha_creacion timestamp
  fecha_actualizacion timestamp
}

Table broker_formatos {
  formato_id int [pk, increment]
  broker_id int
  version varchar(50)
  header_row int
  descripcion text
  archivo_ejemplo varchar(500)
  activo boolean
  fecha_creacion timestamp
  fecha_actualizacion timestamp
}

Ref: broker_formatos.broker_id > brokers.broker_id

Table formato_columnas {
  columna_id int [pk, increment]
  formato_id int
  campo_estandar varchar(100)
  nombre_columna_original varchar(255)
  indice_columna int
  letra_columna varchar(10)
  tipo_dato varchar(50)
  requerido boolean
  descripcion text
  fecha_creacion timestamp
  color_fondo varchar(20)
  color_texto varchar(20)
  es_negrita boolean
  es_cursiva boolean
  tiene_borde boolean
}

Ref: formato_columnas.formato_id > broker_formatos.formato_id

Table broker_metadata {
  metadata_id int [pk, increment]
  formato_id int
  seccion varchar(100)
  campo_nombre varchar(100)
  campo_valor text
  fila_origen int
  columna_origen int
  letra_columna varchar(5)
  fecha_creacion timestamp
}

Ref: broker_metadata.formato_id > broker_formatos.formato_id

Table cotizaciones_archivos {
  archivo_id int [pk, increment]
  broker_id int
  formato_id int
  nombre_archivo varchar(500)
  ruta_archivo text
  vessel_name varchar(255)
  imo_number varchar(50)
  numero_cotizacion varchar(255)
  fecha_archivo date
  fecha_procesado timestamp
}

Ref: cotizaciones_archivos.broker_id > brokers.broker_id
Ref: cotizaciones_archivos.formato_id > broker_formatos.formato_id

Table archivo_colores {
  color_id int [pk, increment]
  archivo_id int
  campo_estandar varchar(100)
  nombre_columna_original varchar(255)
  indice_columna int
  letra_columna varchar(10)
  color_fondo varchar(20)
  color_texto varchar(20)
  es_negrita boolean
  es_cursiva boolean
  tiene_borde boolean
  fecha_deteccion timestamp
}

Ref: archivo_colores.archivo_id > cotizaciones_archivos.archivo_id
